## 一、评估指标体系

### 1.1 核心检索质量指标

| 指标 | 英文 | 定义 | 适用场景 |
|------|------|------|---------|
| Recall@k | Recall@k | 前 k 个结果中命中相关文档的比例 | 评估"是否遗漏" |
| Precision@k | Precision@k | 前 k 个结果中相关文档的比例 | 评估"是否精准" |
| MRR | Mean Reciprocal Rank | 第一个相关结果的倒数排名的均值 | 评估"第一个对的排第几" |
| NDCG@k | Normalized DCG | 考虑排序位置权重的归一化折扣累积增益 | 综合排序质量 |
| Hit@k | Hit Rate | 前 k 个结果至少命中一个相关的比例 | 最宽松指标 |
| MAP | Mean Average Precision | 平均精确率的均值 | 综合精度 |

**建议此项目优先采用**: Recall@5, MRR, Hit@1（最小可行评估集）。

### 1.2 业务特有质量指标

| 指标 | 定义 | 测量方式 |
|------|------|---------|
| 来源可追溯率 | 返回结果中携带 `doc_id`/`rule_id`/`source_text` 的比例 | 检查 SearchHit.entity 字段 |
| 维度匹配准确率 | fact_type / insu_type / hosp_lv / psn_type 等过滤维度匹配的比例 | 对比 query 期望维度 vs 结果维度 |
| 查询改写覆盖率 | question_rewriter 对同义问题的改写一致性 | 构造同义问题对 |
| 降级信号完整性 | retrieval_status 在"无结果"时正确传递 | 检查 SearchQuery.status |

### 1.3 性能指标

| 指标 | 目标 | 测量方式 |
|------|------|---------|
| 向量搜索延迟 | P95 < 100ms | Milvus search() 计时 |
| 端到端检索延迟 | P95 < 500ms | query_understanding → reranker 全链路 |
| Embedding 生成延迟 | P95 < 200ms | embedding_provider.encode() |

### 2.2 标注方法

标注分两级：

1. **宽松标注（自动）**: 利用现有 `policy_rules_v2` 的维度字段（fact_type / hosp_lv / insu_type / psn_type）自动判断相关性。规则：fact_type 匹配 AND 至少 1 个维度字段匹配 → 相关。

2. **严格标注（人工）**: 由医保业务专家对 10-20 个 query 逐条判断返回结果的相关性（relevant / partially_relevant / not_relevant）。

第一阶段先用宽松标注跑通流程，第二阶段补充人工标注。

## 三、评估方法论

### 3.1 评估流程

```
评估 query
  ↓
query_understanding.py  → 生成 SearchQuery（target_object, fact_types, 维度...）
  ↓
Milvus 向量搜索          → 候选集 top_k=10
  ↓
reranker.py              → 重排序
  ↓
评估器 eval_metrics      → 按 expected 标注判断相关性 + 计算指标
  ↓
报告输出
```

关键决策：reranker 是检索管线的一部分，评估时默认开启 reranker。

### 3.2 相关性判断（宽松标注）

```python
def is_relevant(hit: SearchHit, expected: dict) -> bool:
    """宽松标注：fact_type 匹配 + 至少 1 个维度字段匹配"""
    entity = hit.entity or {}

    # fact_type 必须匹配
    if expected.get("fact_types"):
        if entity.get("fact_type") not in expected["fact_types"]:
            return False

    # 至少 1 个维度字段匹配
    dim_fields = ["hosp_lv", "insu_type", "psn_type", "service_type", "admission_order"]
    any_match = False
    for field in dim_fields:
        if field in expected and entity.get(field) == expected[field]:
            any_match = True
            break

    # 如果没有指定维度约束，fact_type 匹配就够
    if not any(f in expected for f in dim_fields):
        return True

    return any_match
```

### 3.3 指标计算参考实现

```python
# eval_metrics.py
def recall_at_k(results: list[SearchHit], expected: dict, k: int = 5) -> float:
    """Recall@k：前 k 个结果中命中相关文档的比例。"""
    total_relevant = expected.get("min_relevant_hits", 1)
    hits_top_k = results[:k]
    relevant_count = sum(1 for hit in hits_top_k if is_relevant(hit, expected))
    return min(relevant_count / total_relevant, 1.0)

def mrr(results: list[SearchHit], expected: dict) -> float:
    """MRR：第一个相关结果的倒数排名。"""
    for rank, hit in enumerate(results, start=1):
        if is_relevant(hit, expected):
            return 1.0 / rank
    return 0.0

def hit_at_k(results: list[SearchHit], expected: dict, k: int = 1) -> float:
    """Hit@k：前 k 个结果中至少命中一个。"""
    for hit in results[:k]:
        if is_relevant(hit, expected):
            return 1.0
    return 0.0

def evaluate_all(queries: list[dict], retriever) -> dict:
    """对全部 query 计算评估指标并汇总。"""
    results = {"per_query": [], "summary": {}}
    for q in queries:
        hits = retriever.search(q["question"], top_k=10)
        results["per_query"].append({
            "id": q["id"],
            "recall@5": recall_at_k(hits, q["expected"], k=5),
            "mrr": mrr(hits, q["expected"]),
            "hit@1": hit_at_k(hits, q["expected"], k=1),
            "result_count": len(hits),
        })

    # 汇总
    summary = results["summary"]
    for metric in ["recall@5", "mrr", "hit@1"]:
        values = [r[metric] for r in results["per_query"]]
        summary[metric] = sum(values) / len(values) if values else 0.0
    return results
```

### 4.2 pytest 集成

```python
# test_retrieval_quality.py
import pytest
import yaml
from pathlib import Path
from src.knowledge_extension.rule_explanation.policy_retrieval.milvus_retriever import MilvusPolicyRetriever
from .eval_metrics import evaluate_all

EVAL_QUERIES_PATH = Path(__file__).parent / "eval_queries.yaml"

@pytest.fixture(scope="module")
def retriever():
    """创建真实的 Milvus retriever（需要 Milvus 运行中）。"""
    try:
        return MilvusPolicyRetriever(host="127.0.0.1", port="19530")
    except Exception as e:
        pytest.skip(f"Milvus 不可用: {e}")

@pytest.fixture(scope="module")
def queries():
    with open(EVAL_QUERIES_PATH, encoding="utf-8") as f:
        return yaml.safe_load(f)["queries"]

class TestRetrievalQuality:
    """检索质量评估 — 需要 Milvus 运行。"""

    def test_recall_at_5_minimum(self, retriever, queries):
        """Recall@5 整体均值 ≥ 0.5（宽松标注下基本门槛）。"""
        results = evaluate_all(queries, retriever)
        avg_recall = results["summary"]["recall@5"]
        assert avg_recall >= 0.50, (
            f"Recall@5 均值 {avg_recall:.2f} 低于门槛 0.50\n"
            f"逐 query: {results['per_query']}"
        )

    def test_mrr_minimum(self, retriever, queries):
        """MRR 整体均值 ≥ 0.3。"""
        results = evaluate_all(queries, retriever)
        avg_mrr = results["summary"]["mrr"]
        assert avg_mrr >= 0.30, f"MRR 均值 {avg_mrr:.2f} 低于门槛 0.30"

    def test_hit_at_1_minimum(self, retriever, queries):
        """Hit@1 整体均值 ≥ 0.4。"""
        results = evaluate_all(queries, retriever)
        avg_hit1 = results["summary"]["hit@1"]
        assert avg_hit1 >= 0.40, f"Hit@1 均值 {avg_hit1:.2f} 低于门槛 0.40"

    def test_no_query_returns_zero(self, retriever, queries):
        """没有 query 返回完全空结果。"""
        for q in queries:
            hits = retriever.search_nodes(q["question"], top_k=10)
            assert len(hits) > 0, f"Q{q['id']} 返回空结果: {q['question']}"

    def test_all_results_have_citations(self, retriever, queries):
        """所有检索结果携带 doc_id/rule_id 来源追溯。"""
        for q in queries:
            hits = retriever.search_facts(q["question"], top_k=10)
            for hit in hits:
                entity = hit.entity or {}
                assert entity.get("policy_id") or entity.get("rule_id"), \
                    f"Q{q['id']} 结果缺少来源追溯: {entity}"
```

### 4.3 回归测试

```python
# test_retrieval_regression.py
import json
import pytest
from pathlib import Path

BASELINE_PATH = Path(__file__).parent / "baseline" / "baseline_v1.json"

class TestRetrievalRegression:
    """与基线对比，防止检索质量退化。"""

    def test_no_regression_vs_baseline(self, retriever, queries):
        if not BASELINE_PATH.exists():
            pytest.skip("基线不存在，首次运行 eval_runner 生成基线后再回归")

        baseline = json.loads(BASELINE_PATH.read_text(encoding="utf-8"))
        current = evaluate_all(queries, retriever)

        for metric in ["recall@5", "mrr", "hit@1"]:
            current_val = current["summary"][metric]
            baseline_val = baseline["summary"].get(metric, 0)
            assert current_val >= baseline_val - 0.05, (
                f"{metric} 退化: baseline={baseline_val:.3f} → current={current_val:.3f}"
            )
```
