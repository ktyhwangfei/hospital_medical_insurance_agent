# 模型与提示词治理控制面 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在不切换现有业务调用的前提下，交付开发/测试环境可用的提示词、模型档案、路由规则前后端管理闭环，并生成明确标记“待接入”的已发布快照。

**Architecture:** 使用一个类型安全的治理资产生命周期处理 `prompt`、`model_profile`、`route_rule` 三类资产：可编辑草稿带 revision，审核后的内容冻结为不可变版本，发布记录维护 dev/test 环境唯一活动版本，回滚产生新发布记录。控制面存储默认 PostgreSQL、显式 `USE_MEMORY_STORAGE=1` 才使用内存；Portal 复用现有 `/model-governance`，不改 `ModelRouter` 和业务调用。

**Tech Stack:** Python 3.12、FastAPI、Pydantic v2、PostgreSQL JSONB、Next.js 16、React 19、TypeScript、Vitest、Playwright。

class PromptAssetContent(BaseModel):
    model_config = ConfigDict(frozen=True)
    asset_type: Literal[GovernanceAssetType.PROMPT] = GovernanceAssetType.PROMPT
    asset_id: str = Field(pattern=r"^[a-z0-9][a-z0-9._-]{2,127}$")
    name: str = Field(min_length=1, max_length=128)
    scene: str = Field(min_length=1, max_length=64)
    model_type: str = "llm"
    system_prompt: str = Field(min_length=1, max_length=20000)
    user_prompt_template: str = Field(min_length=1, max_length=20000)
    variables: list[PromptVariable] = Field(default_factory=list)
    output_mode: Literal["text", "json"] = "text"


GovernanceAssetContent = Annotated[
    PromptAssetContent | ModelProfileAssetContent | RouteRuleAssetContent,
    Field(discriminator="asset_type"),
]


def content_hash(content: GovernanceAssetContent) -> str:
    payload = content.model_dump(mode="json", exclude_none=True)
    raw = json.dumps(payload, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()
```

使用 `string.Formatter().parse()` 提取模板字段；禁止位置字段、属性/索引访问、conversion 和 format spec。模型参数限制为 `temperature 0..2`、`max_tokens 1..65536`，`credential_ref` 仅接受大写环境变量名。路由内容包含 `scene/model_type/profile_id/fallback_profile_ids/enabled`，不得自引用 fallback。

class ModelGovernanceStorage(Protocol):
    def create_draft(self, draft: GovernanceDraft) -> GovernanceDraft: ...
    def update_draft(self, draft: GovernanceDraft, *, expected_revision: int) -> GovernanceDraft: ...
    def get_draft(self, draft_id: str) -> GovernanceDraft | None: ...
    def list_drafts(self, asset_type: GovernanceAssetType | None = None) -> list[GovernanceDraft]: ...
    def save_version(self, version: GovernanceVersion) -> GovernanceVersion: ...
    def get_version(self, version_id: str) -> GovernanceVersion | None: ...
    def list_versions(self, asset_id: str) -> list[GovernanceVersion]: ...
    def save_approval(self, approval: GovernanceApproval) -> GovernanceApproval: ...
    def get_release(self, release_id: str) -> GovernanceRelease | None: ...
    def publish(self, release: GovernanceRelease) -> GovernanceRelease: ...
    def list_releases(self, asset_id: str | None = None) -> list[GovernanceRelease]: ...
    def get_active_release(self, asset_id: str, environment: str) -> GovernanceRelease | None: ...


@lru_cache(maxsize=1)
def get_model_governance_storage() -> ModelGovernanceStorage:
    if os.getenv("USE_MEMORY_STORAGE", "").lower() in {"1", "true", "yes"}:
        return InMemoryModelGovernanceStorage()
    return PostgresModelGovernanceStorage()
```

```python
class ModelGovernancePrincipal(BaseModel):
    user_id: str
    roles: list[str]
    permissions: list[str]


def require_model_governance_permission(required: str, authorization: str | None) -> ModelGovernancePrincipal:
    if os.getenv("MODEL_GOVERNANCE_DEV_MODE", "").lower() not in {"1", "true", "yes"}:
        raise HTTPException(403, detail=error_detail("MODEL_GOVERNANCE_DISABLED", "模型治理写入口仅在显式开发模式开放"))
    auth = authenticator.validate_token(authorization or "")
    permission = authenticator.check_permission(auth, required)
    if not auth.is_success:
        raise HTTPException(401, detail=error_detail("AUTH_INVALID", auth.error_message))
    if not permission.is_success:
        raise HTTPException(403, detail=error_detail("AUTH_FORBIDDEN", permission.error_message))
    return ModelGovernancePrincipal(user_id=auth.user_id, roles=auth.roles, permissions=auth.permissions)
```

写、审核、发布分别要求 `model_governance:write`、`model_governance:review`、`model_governance:publish`。所有成功响应继承 `AgentResponse` 并将 `result` 覆盖为具体 Pydantic 类型；409/422/404 使用 `error_detail()`。
